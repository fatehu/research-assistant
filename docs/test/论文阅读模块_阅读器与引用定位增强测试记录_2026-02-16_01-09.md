```powershell
@'
import asyncio
from app.core.database import async_session_factory
from app.api.literature import _retrieve_rag_sources

async def main():
    async with async_session_factory() as db:
        rows = await _retrieve_rag_sources(
            db,
            knowledge_base_id=107,
            document_ids=[103],
            question='如何提升OCR质量',
            limit=3
        )
        for i, item in enumerate(rows, 1):
            print(i, item.get('page'), item.get('page_source'), item.get('section_title'), item.get('score'))

asyncio.run(main())
'@ | docker compose exec -T backend python -
```

3. 处理：执行 `docker compose exec frontend npm install` 并 `restart frontend`，恢复正常。
